public class Book extends Product{
    public String isbn;
    public String publisher;
    
    public Book(int i, String t, int p, String n, String m){
        super(i, t, p);
        isbn = n;
        publisher = m;
    }
    
    public String printDetail(){
        return getIdTitlePrice() + "\nISBN: "+isbn+" Publisher: "+publisher;
    }
}

public class ComplexNumber extends RealNumber{
    public double com;
    
    public ComplexNumber() {
        super(1.0);
        com = 1.0;
    }
    
    public ComplexNumber(double d1, double d2) {
        super(d1);
        com = d2;
    }
    
    public String details(){
        return getReal() + "\nImaginaryPart: "+com;
    }
}

public class CSEStudent extends Student{
    public String[] crses = {"CSE110", "CSE111", "CSE220", "CSE221"};
    public static int st = 0;
    
    public CSEStudent(String s, int n){
        super(s, n);
        st++;
    }
    
    public static void details(){
        System.out.println("Total CSE Student: "+st);
        System.out.println("Available Lab Based Courses: ");
        System.out.println("CSE110 CSE111 CSE220 CSE221");
    }
    
    public void addLabBasedCourse(String s){
        boolean found = false;
        for(int i = 0; i<4; i++){
            if(crses[i].equals(s)){
                found = true;
                courses += s+" ";
            }
        }
        if(!found){
            System.out.println("It is not a lab based course!");
        }
    }
}
